import httpx
from typing import Dict, Any

class SamsaraClient:
def __init__(self, api_token: str):
self.api_token = api_token
self.base_url = "https://api.samsara.com/v1"

async def get_vehicle_location(self, vehicle_id: str) -> Dict[str, Any]:
headers = {"Authorization": f"Bearer {self.api_token}"}
async with httpx.AsyncClient() as client:
response = await client.get(f"{self.base_url}/fleet/vehicles/{vehicle_id}/stats", headers=headers)
if response.status_code == 200:
data = response.json()
# Parse GPS and HOS
return {
"lat": data.get("gps", {}).get("latitude"),
"lon": data.get("gps", {}).get("longitude"),
"heading": data.get("gps", {}).get("heading"),
"hos_remaining_seconds": data.get("hos", {}).get("drivingSecondsRemaining", 35000)
}
raise ValueError(f"Failed to fetch Samsara telemetry: {response.text}")

import httpx
from typing import Dict, Any

class FmcsaSafetyShield:
def __init__(self):
self.safer_base = "https://mobile.fmcsa.dot.gov/qc/services/carriers"

async def verify_broker_bond(self, mc_or_dot: str) -> Dict[str, Any]:
# Strip leading alpha characters if present
clean_id = ''.join(filter(str.isdigit, mc_or_dot))
async with httpx.AsyncClient() as client:
url = f"{self.safer_base}/{clean_id}"
try:
response = await client.get(url, timeout=10.0)
if response.status_code == 200:
content = response.json()
carrier = content.get("content", {}).get("carrier", {})
is_active = carrier.get("allowedToOperate", "Y") == "Y"
bond_amount = 150000 # Default enforced limit under 2026 rules
return {
"verified": True,
"active": is_active,
"legal_name": carrier.get("legalName", "Unknown Brokerage"),
"bond_amount": bond_amount,

from pydantic_settings import BaseSettings, SettingsConfigDict
from pydantic import Field

class Settings(BaseSettings):
model_config = SettingsConfigDict(env_file=".env", env_file_encoding="utf-8", extra="ignore")

ENVIRONMENT: str = Field(default="production", description="Runtime environment")
TENANT_DATA_DIR: str = Field(default="/app/data/tenants", description="Isolated tenant storage root")

LLM_API_KEY: str = Field(..., description="API key for primary LLM routing")
LLM_MODEL: str = Field(default="meta-llama/llama-3.3-70b-instruct", description="Primary reasoning model")

TELNYX_API_KEY: str = Field(..., description="Telnyx SIP trunking API key")
TELNYX_PHONE_NUMBER: str = Field(..., description="Outbound caller ID phone number")
SIP_WEBSOCKET_URL: str = Field(..., description="Pipecat WebSocket SIP gateway endpoint")

STRIPE_API_KEY: str = Field(..., description="Stripe Connect API secret key for instant split payouts")
DEFAULT_DISPATCH_FEE_PCT: float = Field(default=0.08, description="Default dispatcher cut share (8%)")

FMCSA_API_KEY: str = Field(default="", description="Optional FMCSA data portal API token")

settings = Settings()

C. Core Skill Implementation Modules
1. bona_fide_onboard (src/skills/bona_fide_onboard/handler.py)

Role: Generates 49 CFR § 371.2 legal agreements, dispatches SMS magic links via Telnyx/Twilio, and binds Samsara/Motive ELD OAuth tokens into tenant-isolated storage (~/.hermes/tenants/<tenant_id>/).

2. eld_telemetry (src/skills/eld_telemetry/samsara_client.py)

Role: Polls telematics APIs to stream real-time truck GPS latitude/longitude, heading, and available driving seconds remaining into the agent's context window.

3. safer_bond_shield (src/skills/safer_bond_shield/fmcsa_client.py)

Role: Queries the FMCSA SAFER database using carriers' MC/USDOT numbers to confirm that the bidding broker's surety bond is active ($150K mandate) and checks email header domains against spoofing risks.

4. voice_negotiator (src/skills/voice_negotiator/pipecat_pipeline.py)

Role: Leverages Pipecat (open source) and Telnyx SIP trunks to spin up parallel sub-agent voice pipelines. Places simultaneous outbound calls to brokers, negotiates rates using live HOS telemetry, and surfaces winning bids to human dispatchers via WhatsApp.

5. ratecon_autopsy (src/skills/ratecon_autopsy/vision_parser.py)

Role: Ingests incoming PDF Rate Confirmations, uses multimodal vision models to extract financial line items and accessorial clauses ($50/hr detention terms), and executes binding e-signatures under stored Power-of-Attorney.

6. shopify_split_pay (src/skills/shopify_split_pay/stripe_connect.py)

Role: Connects to Stripe Connect / FedNow clearinghouse rails to execute 1% instant factoring payouts upon BOL scan, automatically executing the mathematical split: 92% to Carrier and 8% to Dispatcher.
